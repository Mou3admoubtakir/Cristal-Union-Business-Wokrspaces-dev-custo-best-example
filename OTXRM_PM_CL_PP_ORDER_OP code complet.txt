METHOD get_properties.

  DATA:
    l_error_msg         TYPE string,
    ls_parent           TYPE /otx/rm_wsci_s_bor,
    ls_child            TYPE /otx/rm_wsci_s_busobj,
    lt_attr_defs        TYPE /otx/rm_wsci_tt_attr_def,
    ls_attr_def         TYPE /otx/rm_wsci_s_attr_def,
    ls_prop             TYPE LINE OF /otx/rm_wsci_tt_properties,
    l_utc_string        TYPE string,
    l_value             TYPE string.

  DATA: ls_oro_key  TYPE /otx/rm_pm_cl_constants=>typ_oro_key,
        ls_ord_key  TYPE /otx/rm_pm_cl_constants=>typ_ord_key,
        ls_tlo_key  TYPE /otx/rm_pm_cl_constants=>typ_tlo_key,
        l_aplzl     TYPE afvgd-aplzl,
        l_arbpl     TYPE rcr01-arbpl.
  DATA: lt_afvgd    TYPE almb_afvgd_tab,
        ls_afvgd    LIKE LINE OF lt_afvgd,
        lt_resbd    TYPE STANDARD TABLE OF resbdget,
        ls_resbd    LIKE LINE OF lt_resbd,
        lt_messages TYPE bal_t_msg,
        ls_messages LIKE LINE OF lt_messages.

  FIELD-SYMBOLS: <ls_bor_properties> TYPE /otx/rm_wsci_s_bor_properties.

  CALL METHOD super->get_properties
    EXPORTING
      pit_bor            = pit_bor
    IMPORTING
      pet_bor_properties = pet_bor_properties.

  CALL METHOD get_property_defs
    IMPORTING
      pet_attr_defs = lt_attr_defs.

*Add properties
  LOOP AT pet_bor_properties ASSIGNING <ls_bor_properties>.
    ls_oro_key = <ls_bor_properties>-bor-id.

*Get the order
    SELECT SINGLE aufnr
      INTO ls_ord_key-aufnr
      FROM afko
      WHERE aufpl = ls_oro_key-aufpl.
    IF NOT sy-subrc IS INITIAL.
      l_error_msg = text-001.
      CREATE OBJECT <ls_bor_properties>-exception
        EXPORTING
          m_error_text = l_error_msg.
      CONTINUE.
    ENDIF.

    CLEAR lt_resbd.
    CALL FUNCTION 'CO_BC_RESBD_OF_ORDER_GET'
      EXPORTING
        aufnr_act = ls_ord_key-aufnr
      TABLES
        resbd_get = lt_resbd.


    CLEAR: lt_afvgd, lt_messages, ls_messages.
    CALL FUNCTION 'IBAPI_ALM_ORDER_OPER_READ'
      EXPORTING
        iv_orderid  = ls_ord_key-aufnr
      IMPORTING
        et_afvgd    = lt_afvgd
      TABLES
        et_messages = lt_messages.

*Check for errors
    CLEAR ls_messages.
    LOOP AT lt_messages INTO ls_messages WHERE msgty CA 'AE'.
      MESSAGE ID ls_messages-msgid TYPE ls_messages-msgty
              NUMBER ls_messages-msgno INTO l_error_msg.
      CREATE OBJECT <ls_bor_properties>-exception
        EXPORTING
          m_error_text = l_error_msg.
      EXIT.
    ENDLOOP.

    IF NOT ls_messages IS INITIAL.
      CONTINUE.
    ENDIF.

    READ TABLE lt_afvgd INTO ls_afvgd
                        WITH KEY aufpl = ls_oro_key-aufpl
                                 aplzl = ls_oro_key-aplzl.
    IF NOT sy-subrc IS INITIAL.
      l_error_msg = text-002.
      CREATE OBJECT <ls_bor_properties>-exception
        EXPORTING
          m_error_text = l_error_msg.
      CONTINUE.
    ENDIF.

    LOOP AT lt_attr_defs INTO ls_attr_def.
      CLEAR ls_prop.
      ls_prop-def = ls_attr_def.

      CASE ls_attr_def-key.
        WHEN c_attr_name_aufpl.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
            EXPORTING
              input  = ls_afvgd-aufpl
            IMPORTING
              output = l_value.
          CONDENSE l_value.
          APPEND l_value TO ls_prop-values.

        WHEN c_attr_name_aplzl.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
            EXPORTING
              input  = ls_afvgd-aplzl
            IMPORTING
              output = l_value.
          CONDENSE l_value.
          APPEND l_value TO ls_prop-values.

        WHEN c_attr_name_plnkn.
          IF NOT ls_afvgd-plnkn IS INITIAL.
            CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
              EXPORTING
                input  = ls_afvgd-plnkn
              IMPORTING
                output = l_value.
            CONDENSE l_value.
            APPEND l_value TO ls_prop-values.
          ENDIF.

        WHEN c_attr_name_plnal.
          APPEND ls_afvgd-plnal TO ls_prop-values.

        WHEN c_attr_name_plnty.
          APPEND ls_afvgd-plnty TO ls_prop-values.

        WHEN c_attr_name_plnnr.
          IF NOT ls_afvgd-plnnr IS INITIAL.
            CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
              EXPORTING
                input  = ls_afvgd-plnnr
              IMPORTING
                output = l_value.
            CONDENSE l_value.
            APPEND l_value TO ls_prop-values.
          ENDIF.

        WHEN c_attr_name_vornr.
          IF ls_afvgd-sumnr IS INITIAL.
            APPEND ls_afvgd-vornr TO ls_prop-values.
          ELSE.
            APPEND ls_afvgd-uvorn TO ls_prop-values.
          ENDIF.

        WHEN c_attr_name_steus.
          APPEND ls_afvgd-steus TO ls_prop-values.

        WHEN c_attr_name_arbid.
          CALL FUNCTION 'CR_WORKSTATION_READ'
            EXPORTING
              id        = ls_afvgd-arbid
            IMPORTING
              arbpl     = l_arbpl
            EXCEPTIONS
              not_found = 1
              OTHERS    = 2.
          IF sy-subrc <> 0.
            l_error_msg = text-003.
            CREATE OBJECT <ls_bor_properties>-exception
              EXPORTING
                m_error_text = l_error_msg.
            CONTINUE.
          ENDIF.
          APPEND l_arbpl TO ls_prop-values.
          "APPEND ls_afvgd-arbid TO ls_prop-values.

        WHEN c_attr_name_werks.
          APPEND ls_afvgd-werks TO ls_prop-values.

        WHEN c_attr_name_ltxa1.
          APPEND ls_afvgd-ltxa1 TO ls_prop-values.

        WHEN c_attr_name_sumnr.
          APPEND ls_afvgd-sumnr TO ls_prop-values.

        WHEN c_attr_name_usr00.
          APPEND ls_afvgd-usr00 TO ls_prop-values.

        WHEN c_attr_name_usr01.
          APPEND ls_afvgd-usr01 TO ls_prop-values.

        WHEN c_attr_name_usr02.
          APPEND ls_afvgd-usr02 TO ls_prop-values.

        WHEN c_attr_name_usr03.
          APPEND ls_afvgd-usr03 TO ls_prop-values.

        WHEN c_attr_name_usr08.
          l_utc_string = convert_dats_to_string( ls_afvgd-usr08 ).
          APPEND l_utc_string TO ls_prop-values.

        WHEN c_attr_name_usr09.
          l_utc_string = convert_dats_to_string( ls_afvgd-usr09 ).
          APPEND l_utc_string TO ls_prop-values.

        WHEN c_attr_name_usr10.
          APPEND ls_afvgd-usr10 TO ls_prop-values.

        WHEN c_attr_name_usr11.
          APPEND ls_afvgd-usr11 TO ls_prop-values.

        WHEN OTHERS.
          " Maybe this property was already returned
          " by the parent method.
          CONTINUE.
      ENDCASE.

      APPEND ls_prop TO <ls_bor_properties>-properties.
    ENDLOOP.

*Add composite parents
    CLEAR ls_parent.
    IF ls_afvgd-sumnr IS INITIAL.
      ls_parent-type = /otx/rm_pm_cl_constants=>c_bo_order_super.
      ls_parent-id   = ls_ord_key.
      <ls_bor_properties>-composite_parent-type = ls_parent-type.
      <ls_bor_properties>-composite_parent-id   = ls_parent-id.
    ELSE.
*It is a suboperation
      ls_parent-type = /otx/rm_pm_cl_constants=>c_bo_order_operation_super.
      ls_oro_key-aplzl = ls_afvgd-sumnr.
      ls_parent-id = ls_oro_key.
      <ls_bor_properties>-composite_parent-type = ls_parent-type.
      <ls_bor_properties>-composite_parent-id   = ls_parent-id.
    ENDIF.

*Add other relations
    IF NOT ls_afvgd-plnnr IS INITIAL.
      CLEAR ls_parent.
      ls_parent-type = /otx/rm_pm_cl_constants=>c_bo_tasklist_operation.
      ls_tlo_key-plnty = ls_afvgd-plnty.
      ls_tlo_key-plnnr = ls_afvgd-plnnr.
      ls_tlo_key-plnkn = ls_afvgd-plnkn.
      ls_tlo_key-zaehl = ls_afvgd-zaehl.
      ls_parent-id = ls_tlo_key.
      APPEND ls_parent TO <ls_bor_properties>-parents.
    ENDIF.

*Add composite children
    l_aplzl = ls_afvgd-aplzl.
    LOOP AT lt_afvgd INTO ls_afvgd WHERE sumnr = l_aplzl.
      CLEAR ls_child.
      ls_child-type = /otx/rm_pm_cl_constants=>c_bo_order_operation_super.
      ls_oro_key-aufpl = ls_afvgd-aufpl.
      ls_oro_key-aplzl = ls_afvgd-aplzl.
      ls_child-id = ls_oro_key.
      APPEND ls_child TO <ls_bor_properties>-composite_children.
    ENDLOOP.

*Add material relation
    LOOP AT lt_resbd INTO ls_resbd WHERE aufpl = ls_oro_key-aufpl
                                     AND aplzl = ls_oro_key-aplzl.
      ls_parent-type = /otx/rm_pm_cl_constants=>c_bo_material_super.
      ls_parent-id = ls_resbd-matnr.
      APPEND ls_parent TO <ls_bor_properties>-parents.
    ENDLOOP.

  ENDLOOP.
ENDMETHOD.
