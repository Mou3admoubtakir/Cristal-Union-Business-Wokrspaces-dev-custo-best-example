METHOD GET_PROPERTIES.

  DATA:
    l_error_msg         TYPE string,
    l_id                TYPE bapi2080_nothdre-notif_no,
    ls_notifheader      TYPE bapi2080_nothdre,
    ls_notifhdtext      TYPE bapi2080_nothdtxte,
    ls_notif_type       TYPE bapi2080_nothdre-notif_type,
    lt_return           TYPE TABLE OF bapiret2,
    ls_return           TYPE bapiret2,
    ls_parent           TYPE /otx/rm_wsci_s_bor,
    ls_child            TYPE /otx/rm_wsci_s_bor,
    lt_attr_defs        TYPE /otx/rm_wsci_tt_attr_def,
    ls_attr_def         TYPE /otx/rm_wsci_s_attr_def,
    ls_prop             TYPE LINE OF /otx/rm_wsci_tt_properties,
    l_value             TYPE string,
    ls_rcr01            TYPE rcr01,
    l_tplnr             TYPE tplnr,

    l_equnr             TYPE equnr.

  FIELD-SYMBOLS:
    <ls_bor_properties> TYPE /otx/rm_wsci_s_bor_properties.


  CALL METHOD super->get_properties
    EXPORTING
      pit_bor            = pit_bor
    IMPORTING
      pet_bor_properties = pet_bor_properties.

  CALL METHOD get_property_defs
    IMPORTING
      pet_attr_defs = lt_attr_defs.

  " Goes through each given BOR.
  LOOP AT pet_bor_properties ASSIGNING <ls_bor_properties>.

    " Gets the details.
    l_id = <ls_bor_properties>-bor-id.
    CLEAR: ls_notifheader, ls_notifhdtext,
           lt_return, ls_return.
    CALL FUNCTION 'BAPI_ALM_NOTIF_GET_DETAIL'
      EXPORTING
        number             = l_id
      IMPORTING
        notifheader_export = ls_notifheader
        notifhdtext        = ls_notifhdtext
      TABLES
        return             = lt_return.

    " Error handling
    LOOP AT lt_return INTO ls_return WHERE type EQ 'E' OR type EQ 'A'.
      EXIT.
    ENDLOOP.

*    IF ls_return IS NOT INITIAL.
*      l_error_msg = ls_return-message.
*      CREATE OBJECT <ls_bor_properties>-exception
*        EXPORTING
*          m_error_text = l_error_msg.
*      CONTINUE.
*    ENDIF.
   IF ls_notifheader-NOTIF_TYPE NE ''. "Cas de la création du WS durant la création de l'avis.
       CLEAR: ls_notif_type.
       SELECT SINGLE value FROM ZAEROW_XECMPARAM
       INTO  ls_notif_type
       WHERE sysid = sy-sysid
       AND   type  = 'W'
       AND   object = 'BUS2038'
       AND   param = 'NOTIF_TYPE'
       AND   value = ls_notifheader-NOTIF_TYPE.

      IF ls_notif_type is initial.
        CREATE OBJECT <ls_bor_properties>-exception
        TYPE
          /OTX/CX_RM_WSCI_NO_DECLARING
        EXPORTING
              m_error_text = 'Type de notification/avis non pris en charge'
              M_LONG_ERROR_TEXT = 'Type de notification/avis non pris en charge'.
        CONTINUE.
     ENDIF.
   ENDIF.





    " Sets the parent equipment or functional location, if any.
    IF ls_notifheader-equipment IS NOT INITIAL.
      ls_parent-id   = ls_notifheader-equipment.
      ls_parent-type = /otx/rm_pm_cl_constants=>c_bo_equi_super.
      APPEND ls_parent TO <ls_bor_properties>-parents.
    ENDIF.
    IF ls_notifheader-funct_loc IS NOT INITIAL.
      ls_parent-id   = ls_notifheader-funct_loc.
      ls_parent-type = /otx/rm_pm_cl_constants=>c_bo_func_loc_super.
      APPEND ls_parent TO <ls_bor_properties>-parents.
    ENDIF.
    IF ls_notifheader-orderid IS NOT INITIAL.
      ls_child-id   = ls_notifheader-orderid.
      ls_child-type = /otx/rm_pm_cl_constants=>c_bo_order_super.
      APPEND ls_child TO <ls_bor_properties>-children.
    ENDIF.

    " Gets the properties.
    LOOP AT lt_attr_defs INTO ls_attr_def.

      CLEAR ls_prop.
      ls_prop-def = ls_attr_def.

      CASE ls_attr_def-key.

        WHEN c_attr_name_notif_dat.
          l_value = convert_datstimstz_to_string(
              pi_dats  = ls_notifheader-notif_date
              pi_tims  = ls_notifheader-notiftime
              pi_tzone = ls_notifheader-notiftmez
              pi_append_time_diff = abap_true ).
          APPEND l_value TO ls_prop-values.

        WHEN c_attr_name_notif_type.
          APPEND ls_notifheader-notif_type TO ls_prop-values.

        WHEN c_attr_name_maintenance_plant.
          APPEND ls_notifheader-planplant TO ls_prop-values.

        WHEN c_attr_name_equipment.
          WRITE ls_notifheader-equipment TO l_equnr.
          APPEND l_equnr TO ls_prop-values.

        WHEN c_attr_name_equipment_desc.
          APPEND ls_notifhdtext-equidescr TO ls_prop-values.

        WHEN c_attr_name_func_location.
          WRITE ls_notifheader-funct_loc TO l_tplnr.
          APPEND l_tplnr TO ls_prop-values.

        WHEN c_attr_name_func_location_desc.
          APPEND ls_notifhdtext-funcldescr TO ls_prop-values.

        WHEN c_attr_name_desc.
          APPEND ls_notifhdtext-cntr_descr TO ls_prop-values.

        WHEN c_attr_name_short_text.
          APPEND ls_notifheader-short_text TO ls_prop-values.

        WHEN c_attr_name_coding.
          APPEND ls_notifheader-coding TO ls_prop-values.

        WHEN c_attr_name_pm_wkctr.
          IF NOT ls_notifheader-pm_wkctr IS INITIAL.
            CALL FUNCTION 'CR_WORKSTATION_READ'
              EXPORTING
                id        = ls_notifheader-pm_wkctr
              IMPORTING
                arbpl     = ls_rcr01-arbpl
              EXCEPTIONS
                not_found = 1
                OTHERS    = 2.
            IF sy-subrc <> 0.
              l_error_msg =  'Could not find the work center for' &
                 'ARBID = &1'.                              "#EC NOTEXT
              REPLACE '&1' IN l_error_msg WITH ls_notifheader-pm_wkctr.
              CREATE OBJECT <ls_bor_properties>-exception
                EXPORTING
                  m_error_text = l_error_msg.
              EXIT.
            ENDIF.
            APPEND ls_rcr01-arbpl TO ls_prop-values.
          ENDIF.

        WHEN c_attr_name_sortfield.
          APPEND ls_notifheader-sortfield TO ls_prop-values.

        WHEN c_attr_name_abcindic.
          APPEND ls_notifheader-abcindic TO ls_prop-values.

        WHEN c_attr_name_created_by.
          APPEND ls_notifheader-created_by TO ls_prop-values.

        WHEN c_attr_name_reportedby.
          APPEND ls_notifheader-reportedby TO ls_prop-values.

        WHEN c_attr_name_notif_no.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
            EXPORTING
              input  = l_id
            IMPORTING
              output = l_value.
          CONDENSE l_value.
          APPEND l_value TO ls_prop-values.

        WHEN OTHERS.
          " Maybe this property was already returned
          " by the parent method.
          CONTINUE.

      ENDCASE.

      APPEND ls_prop TO <ls_bor_properties>-properties.

    ENDLOOP.

  ENDLOOP.

ENDMETHOD.
