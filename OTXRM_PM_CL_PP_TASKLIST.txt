METHOD get_properties.

  DATA:
    ls_child            TYPE /otx/rm_wsci_s_busobj,
    ls_parent           TYPE /otx/rm_wsci_s_bor,
    lt_attr_defs        TYPE /otx/rm_wsci_tt_attr_def,
    ls_attr_def         TYPE /otx/rm_wsci_s_attr_def,
    l_error_text        TYPE string,
    ls_prop             TYPE LINE OF /otx/rm_wsci_tt_properties.

  DATA: ls_tli_key  TYPE /otx/rm_pm_cl_constants=>typ_tli_key,
        ls_tlo_key  TYPE /otx/rm_pm_cl_constants=>typ_tlo_key.

  DATA: ls_plko     TYPE plko,
        ls_plas     TYPE plas,
        lt_plas     TYPE STANDARD TABLE OF plas,
        ls_rcr01    TYPE rcr01,
        l_value     TYPE string.

  FIELD-SYMBOLS: <ls_bor_properties> TYPE /otx/rm_wsci_s_bor_properties.

  CALL METHOD super->get_properties
    EXPORTING
      pit_bor            = pit_bor
    IMPORTING
      pet_bor_properties = pet_bor_properties.

  CALL METHOD get_property_defs
    IMPORTING
      pet_attr_defs = lt_attr_defs.

*Add properties
  LOOP AT pet_bor_properties ASSIGNING <ls_bor_properties>.
    ls_tli_key = <ls_bor_properties>-bor-id.

    SELECT SINGLE * INTO ls_plko
      FROM plko
     WHERE plnty = ls_tli_key-plnty
       AND plnnr = ls_tli_key-plnnr
       AND plnal = ls_tli_key-plnal.
    IF sy-subrc <> 0.
      l_error_text =  'Could not find task list in table plko ' &
         'with key plnty = &1, plnnr =  &2, plnal = &3'.    "#EC NOTEXT
      REPLACE '&1' IN l_error_text WITH ls_tli_key-plnty.
      REPLACE '&2' IN l_error_text WITH ls_tli_key-plnnr.
      REPLACE '&3' IN l_error_text WITH ls_tli_key-plnal.
      CREATE OBJECT <ls_bor_properties>-exception
        EXPORTING
          m_error_text = l_error_text.
      CONTINUE.
    ENDIF.

    LOOP AT lt_attr_defs INTO ls_attr_def.
      CLEAR ls_prop.
      ls_prop-def = ls_attr_def.

      CASE ls_attr_def-key.
        WHEN c_attr_name_ktext.
          APPEND ls_plko-ktext TO ls_prop-values.

        WHEN c_attr_name_iwerk.
          APPEND ls_plko-iwerk TO ls_prop-values.

        WHEN c_attr_name_arbid.
          IF NOT ls_plko-arbid IS INITIAL.
            CALL FUNCTION 'CR_WORKSTATION_READ'
              EXPORTING
                id        = ls_plko-arbid
              IMPORTING
                arbpl     = ls_rcr01-arbpl
              EXCEPTIONS
                not_found = 1
                OTHERS    = 2.
            IF sy-subrc <> 0.
              l_error_text =  'Could not find the work center for' &
                 'ARBID = &1'.                              "#EC NOTEXT
              REPLACE '&1' IN l_error_text WITH ls_plko-arbid.
              CREATE OBJECT <ls_bor_properties>-exception
                EXPORTING
                  m_error_text = l_error_text.
              EXIT.
            ENDIF.
            APPEND ls_rcr01-arbpl TO ls_prop-values.
          ENDIF.

        WHEN c_attr_name_plnal.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
            EXPORTING
              input  = ls_plko-plnal
            IMPORTING
              output = l_value.
          CONDENSE l_value.
          APPEND l_value TO ls_prop-values.

        WHEN c_attr_name_plnnr.
          CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
            EXPORTING
              input  = ls_plko-plnnr
            IMPORTING
              output = l_value.
          CONDENSE l_value.
          APPEND l_value TO ls_prop-values.

        WHEN c_attr_name_plnty.
          APPEND ls_plko-plnty TO ls_prop-values.

        WHEN c_attr_name_zaehl.
          APPEND ls_plko-zaehl TO ls_prop-values.

        WHEN c_attr_name_werks.
          APPEND ls_plko-werks TO ls_prop-values.

        WHEN c_attr_name_istru.
          APPEND ls_plko-istru TO ls_prop-values.

        WHEN c_attr_name_verwe.
          APPEND ls_plko-verwe TO ls_prop-values.

        WHEN c_attr_name_vagrp.
          APPEND ls_plko-vagrp TO ls_prop-values.

        WHEN c_attr_name_strat.
          APPEND ls_plko-strat TO ls_prop-values.

        WHEN c_attr_name_loekz.
          IF NOT ls_plko-loekz IS INITIAL.
            APPEND '!' TO ls_prop-values.
          ELSE.
            APPEND ls_plko-loekz TO ls_prop-values.
          ENDIF.

        WHEN OTHERS.
          " Maybe this property was already returned
          " by the parent method.
          CONTINUE.
      ENDCASE.

      APPEND ls_prop TO <ls_bor_properties>-properties.
    ENDLOOP.

*Add operations
    SELECT plnty plnnr plnal plnkn FROM plas
      INTO CORRESPONDING FIELDS OF TABLE lt_plas
      WHERE plnty = ls_tli_key-plnty
        AND plnnr = ls_tli_key-plnnr
        AND plnal = ls_tli_key-plnal.
    LOOP AT lt_plas INTO ls_plas.
      ls_child-type = /otx/rm_pm_cl_constants=>c_bo_tasklist_operation.
      ls_tlo_key-plnty = ls_plas-plnty.
      ls_tlo_key-plnnr = ls_plas-plnnr.
      ls_tlo_key-plnkn = ls_plas-plnkn.
      ls_tlo_key-zaehl = ls_plas-plnkn.
      ls_child-id   = ls_tlo_key.
      APPEND ls_child TO <ls_bor_properties>-composite_children.
    ENDLOOP.

*Add material relations
    IF NOT ls_plko-istru IS INITIAL.
      ls_parent-type = /otx/rm_pm_cl_constants=>c_bo_material_super.
      ls_parent-id = ls_plko-istru.
      APPEND ls_parent TO <ls_bor_properties>-parents.
    ENDIF.
  ENDLOOP.
ENDMETHOD.
