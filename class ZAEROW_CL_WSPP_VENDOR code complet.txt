* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Public Method ZAEROW_CL_WSPP_VENDOR->GET_DATA
* +-------------------------------------------------------------------------------------------------+
* | [--->] PIS_BOR                        TYPE        /OTX/RM_WSCI_S_BOR
* | [--->] PI_STRUCT_NAME                 TYPE        TABNAME
* | [<---] PET_COMPOSITE_CHILDREN         TYPE        /OTX/RM_WSCI_TT_BUSOBJ
* | [<---] PES_COMPOSITE_PARENT           TYPE        /OTX/RM_WSCI_S_BUSOBJ
* | [<---] PET_ROLES_USERS                TYPE        /OTX/RM_WSCI_TT_ROLE_USERS
* | [<---] PET_CHILDREN_EXT               TYPE        /OTX/RM_WSCI_TT_BUSOBJ
* | [<---] PET_PARENTS_EXT                TYPE        /OTX/RM_WSCI_TT_BUSOBJ
* | [<---] PET_IDENTICAL_BUSINESS_OBJECTS TYPE        /OTX/RM_WSCI_TT_BUSOBJ
* | [<---] PE_DATA                        TYPE        ANY
* | [<---] PET_PARENTS                    TYPE        /OTX/RM_WSCI_TT_BOR
* | [<---] PET_CHILDREN                   TYPE        /OTX/RM_WSCI_TT_BOR
* | [<---] PEH_EXCEPTION                  TYPE REF TO /OTX/CX_RM_WSCI_WSPROVIDER
* | [!CX!] /OTX/CX_RM_WSCI_WSPROVIDER
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_DATA.

  DATA:
    l_vendor_number         TYPE lifnr,
    l_vendor_number_display TYPE lifnr,
    ls_vendor               TYPE lfa1,
    lt_purch_org_data       TYPE TABLE OF lfm1,
    ls_purch_org_data       TYPE lfm1,
    l_bus_partner_guid      TYPE bu_partner_guid,
    l_bus_partner_number    TYPE bu_partner,
    l_landx                 TYPE t005t-landx,
    lh_bp_vendor            TYPE REF TO cvi_ka_bp_vendor,
    ls_t024e                TYPE t024e,
    l_rc                    TYPE sy-subrc,
    ls_relation             TYPE /otx/rm_wsci_s_busobj,
    ls_wsci_bor             TYPE /otx/rm_wsci_s_bor,
    l_long_error_text       TYPE string,
    l_ext_sys               TYPE string,
    l_error_text            TYPE string.

    CLEAR me->ms_data.

    " Get the details of vendor
    l_vendor_number = pis_bor-id.
    CLEAR ls_vendor.
    CALL FUNCTION 'WY_LFA1_SINGLE_READ'
      EXPORTING
        pi_lifnr         = l_vendor_number
      IMPORTING
        po_lfa1          = ls_vendor
      EXCEPTIONS
        no_records_found = 1
        internal_error   = 2
        OTHERS           = 3.

    IF ls_vendor-name3 IS NOT INITIAL.
      "#CONCATENATE ls_vendor-name1 '(' ls_vendor-name3 ')'  INTO ls_vendor-name1.
    ENDIF.
    IF sy-subrc NE 0.
      IF sy-msgid IS NOT INITIAL.
        MESSAGE ID sy-msgid TYPE sy-msgty NUMBER sy-msgno
                WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4
                INTO l_long_error_text.
      ENDIF.

      l_error_text = 'Could not read data of Vendor with number ='. "#EC NOTEXT
      CONCATENATE l_error_text ' ' pis_bor-id ':' l_long_error_text INTO l_error_text RESPECTING BLANKS.
      RAISE EXCEPTION TYPE /otx/cx_rm_wsci_wsprovider
        EXPORTING
          m_error_text = l_error_text.
    ENDIF.

    MOVE-CORRESPONDING ls_vendor TO me->ms_data.

    " Localize country of vendor
     CALL FUNCTION 'BKK_BKKA_COUNTRY_GET'
      EXPORTING
        i_land1       = ls_vendor-land1
     IMPORTING
       e_landx       = l_landx
       e_rc          = l_rc.

    IF l_rc <> 0.
      l_error_text = 'Could not localize country for country ='. "#EC NOTEXT
      CONCATENATE l_error_text ' ' ls_vendor-land1 INTO l_error_text RESPECTING BLANKS.
      RAISE EXCEPTION TYPE /otx/cx_rm_wsci_wsprovider
      EXPORTING
        m_error_text = l_error_text.
    ENDIF.
    me->ms_data-landx = l_landx.

    " Get purchasing organization data
    CLEAR lt_purch_org_data.
    CALL FUNCTION 'LFM1_GENERIC_READ_LIFNR'
      EXPORTING
        i_lifnr   = l_vendor_number
      TABLES
        o_lfm1    = lt_purch_org_data
      EXCEPTIONS
        not_found = 1 ##FM_SUBRC_OK
        OTHERS    = 2.


   READ TABLE lt_purch_org_data INTO ls_purch_org_data INDEX 1.
   me->ms_data-purorg = ls_purch_org_data-ekorg.

   IF ls_purch_org_data-ekorg IS NOT INITIAL.
    CLEAR ls_t024e.
    CALL FUNCTION 'T024E_SINGLE_READ'
         EXPORTING
          kzrfb             = ' '
           t024e_ekorg      = ls_purch_org_data-ekorg
        IMPORTING
          wt024e            = ls_t024e
        EXCEPTIONS
          not_found         = 1
          OTHERS            = 2.
   ENDIF.

   IF ls_t024e IS NOT INITIAL AND sy-subrc NE 1.
      me->ms_data-purorg_txt = ls_t024e-ekotx.
   ENDIF.

    " Remove trailing zeros from vendor number
   CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        input  = l_vendor_number
      IMPORTING
        output = l_vendor_number_display.
     me->ms_data-lifnr = l_vendor_number_display.


    " Try to get assigned business partner
    "lh_bp_vendor = cvi_ka_bp_vendor=>get_instance( ).
    "l_bus_partner_guid = lh_bp_vendor->get_assigned_bp_for_vendor( i_vendor = l_vendor_number ).

   IF l_bus_partner_guid IS NOT INITIAL.
      l_bus_partner_number = lh_bp_vendor->get_partner_id( i_partner_guid = l_bus_partner_guid ).
   ENDIF.

   " Remove trailing zeros from number
   CALL FUNCTION 'CONVERSION_EXIT_ALPHA_OUTPUT'
      EXPORTING
        input  = l_bus_partner_number
      IMPORTING
        output = l_bus_partner_number.

   "  Append Business Partner as parent
   IF  l_bus_partner_number IS NOT INITIAL.
      ls_relation-id                  = l_bus_partner_number.
      ls_relation-type                = me->co_bo_type_bupa.
      " Get external sytem id
      l_ext_sys = me->get_ext_sys_from_bo_type( me->co_bo_type_bupa ).
      ls_relation-external_system_id = l_ext_sys.
      APPEND ls_relation TO pet_children_ext.

      ls_wsci_bor-id    = l_bus_partner_number.
      ls_wsci_bor-type  = me->co_bo_type_bupa.
      APPEND ls_wsci_bor TO pet_children.
   ENDIF.




   pe_data = me->ms_data.

  ENDMETHOD.


* <SIGNATURE>---------------------------------------------------------------------------------------+
* | Instance Private Method ZAEROW_CL_WSPP_VENDOR->GET_EXT_SYS_FROM_BO_TYPE
* +-------------------------------------------------------------------------------------------------+
* | [--->] PI_BO_TYPE                     TYPE        STRING
* | [<-()] PR_EXTERNAL_SYSTEM_ID          TYPE        STRING
* | [!CX!] /OTX/CX_RM_WSCI_WSPROVIDER
* +--------------------------------------------------------------------------------------</SIGNATURE>
  METHOD GET_EXT_SYS_FROM_BO_TYPE.

  DATA:
  ls_elib_info           TYPE  /otx/rm_telib,
  lv_elib_with_quotes    TYPE string,
  lv_error_message       TYPE string,
  l_wsci_bor_type        TYPE swo_objtyp,
  lv_elib_id             TYPE /otx/rmcon_de_les_id,
  ls_cx_rm_wdgen         TYPE REF TO /otx/cx_rm_wdgen.

  l_wsci_bor_type = pi_bo_type.

  " Get elib ID
  TRY.
    lv_elib_id = /otx/rm_wdgen_widget_config=>get_elib_id_from_bo_type( l_wsci_bor_type ).
  CATCH /otx/cx_rm_wdgen INTO ls_cx_rm_wdgen.
    lv_error_message = ls_cx_rm_wdgen->get_text( ).
    RAISE EXCEPTION TYPE /otx/cx_rm_wsci_wsprovider
      EXPORTING m_error_text = lv_error_message.
  ENDTRY.

  " Get external sytem ID
  SELECT SINGLE * FROM /otx/rm_telib INTO ls_elib_info
    WHERE les_id = lv_elib_id.
  IF sy-subrc <> 0.
    lv_error_message = text-001. " elib not found
    CONCATENATE '"' lv_elib_id '"' INTO lv_elib_with_quotes. "#EC NOTEXT
    REPLACE ALL OCCURRENCES OF '$ELIB' "#EC NOTEXT
      IN lv_error_message WITH lv_elib_with_quotes.
    RAISE EXCEPTION TYPE /otx/cx_rm_wsci_wsprovider
      EXPORTING m_error_text = lv_error_message.
  ENDIF.

  pr_external_system_id = ls_elib_info-external_sys_id.

  ENDMETHOD.
